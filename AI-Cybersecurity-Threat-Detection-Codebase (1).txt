"""
File: data/generate_dataset.py
Description: Generates a realistic network traffic dataset simulating normal corporate

             Port Scan, and Zero-Day Exfiltration Anomaly).
Author: AI-Powered Cybersecurity Threat Detection Project

def generate_cybersecurity_dataset(num_samples: int = 2500, random_state: int = 42) -> pd.DataFrame:

    """
    np.random.seed(random_state)

    # Distribution: 65% Normal, 14% DoS, 10% Brute Force, 7% Port Scan, 4% Exfiltration Anomaly
    categories = np.random.choice(
        ["Normal", "SYN_Flood_DoS", "SSH_Brute_Force", "Port_Scan", "Data_Exfiltration"],
        size=num_samples,
        p=[0.65, 0.14, 0.10, 0.07, 0.04],
    )

    records = []
    base_timestamp = pd.Timestamp("2026-10-10 08:00:00")

    for i, category in enumerate(categories):
        timestamp = base_timestamp + pd.Timedelta(seconds=i * np.random.randint(1, 5))

        if category == "Normal":
            duration = np.random.exponential(scale=12.0)
            protocol_type = np.random.choice(["TCP", "UDP", "ICMP"], p=[0.78, 0.19, 0.03])
            service = np.random.choice(["HTTPS", "HTTP", "DNS", "SSH", "SMTP"], p=[0.50, 0.25, 0.15, 0.05, 0.05])
            src_bytes = int(np.random.lognormal(mean=7.2, sigma=0.8))
            dst_bytes = int(np.random.lognormal(mean=8.0, sigma=0.9))
            packet_rate = round(np.random.uniform(10, 180), 2)
            syn_error_rate = round(np.random.uniform(0.0, 0.06), 3)
            failed_logins = int(np.random.choice([0, 1], p=[0.97, 0.03]))
            dst_port_diversity = int(np.random.randint(1, 4))
            is_threat = 0

        elif category == "SYN_Flood_DoS":
            duration = np.random.uniform(0.01, 1.5)
            protocol_type = "TCP"
            service = np.random.choice(["HTTP", "HTTPS"], p=[0.7, 0.3])
            src_bytes = int(np.random.uniform(40, 160))
            dst_bytes = int(np.random.uniform(0, 60))
            packet_rate = round(np.random.uniform(1800, 5200), 2)
            syn_error_rate = round(np.random.uniform(0.78, 0.99), 3)
            failed_logins = 0
            dst_port_diversity = int(np.random.randint(1, 3))
            is_threat = 1

        elif category == "SSH_Brute_Force":
            duration = np.random.uniform(3.0, 45.0)
            protocol_type = "TCP"
            service = "SSH"
            src_bytes = int(np.random.uniform(600, 3200))
            dst_bytes = int(np.random.uniform(400, 1800))
            packet_rate = round(np.random.uniform(85, 420), 2)
            syn_error_rate = round(np.random.uniform(0.05, 0.25), 3)
            failed_logins = int(np.random.randint(6, 38))
            dst_port_diversity = 1
            is_threat = 1

        elif category == "Port_Scan":
            duration = np.random.uniform(0.05, 2.0)
            protocol_type = np.random.choice(["TCP", "UDP"], p=[0.85, 0.15])
            service = np.random.choice(["HTTP", "SSH", "SMTP", "DNS"])
            src_bytes = int(np.random.uniform(64, 256))
            dst_bytes = int(np.random.uniform(0, 120))
            packet_rate = round(np.random.uniform(450, 1400), 2)
            syn_error_rate = round(np.random.uniform(0.45, 0.85), 3)
            failed_logins = int(np.random.randint(0, 2))
            dst_port_diversity = int(np.random.randint(25, 120))
            is_threat = 1

        else:  # Data_Exfiltration (Zero-Day Anomaly profile)
            duration = np.random.uniform(40.0, 240.0)
            protocol_type = np.random.choice(["TCP", "UDP"], p=[0.6, 0.4])
            service = np.random.choice(["HTTPS", "DNS"], p=[0.65, 0.35])
            src_bytes = int(np.random.uniform(250000, 1200000))
            dst_bytes = int(np.random.uniform(800, 4500))
            packet_rate = round(np.random.uniform(320, 950), 2)
            syn_error_rate = round(np.random.uniform(0.01, 0.10), 3)
            failed_logins = int(np.random.randint(0, 3))
            dst_port_diversity = int(np.random.randint(2, 8))
            is_threat = 1

        records.append({
            "flow_id": f"FLW-{10000 + i}",
            "timestamp": timestamp.strftime("%Y-%m-%d %H:%M:%S"),

"""
File: src/preprocess.py
Description: Data cleaning, categorical encoding, and feature scaling module
             for network traffic datasets.
"""

import os
import joblib
import pandas as pd
from sklearn.preprocessing import LabelEncoder, StandardScaler


CATEGORICAL_COLS = ["protocol_type", "service"]


def clean_and_preprocess_data(df: pd.DataFrame) -> pd.DataFrame:
    """
    Cleans raw network flow records:
    1. Drops duplicate flow records
    2. Fills any missing numeric values with median
    3. Encodes categorical columns (protocol_type, service)
    """
    print("[INFO] Starting data cleaning and preprocessing...")
    initial_rows = len(df)

    # 1. Remove duplicate rows
    df = df.drop_duplicates().copy()

    # 2. Handle missing values in numeric columns
    numeric_cols = df.select_dtypes(include=["float64", "int64"]).columns
    for col in numeric_cols:
        if df[col].isnull().sum() > 0:
            df[col] = df[col].fillna(df[col].median())

    # 3. Encode categorical columns
    encoders = {}
    for col in CATEGORICAL_COLS:
        le = LabelEncoder()
        df[f"{col}_encoded"] = le.fit_transform(df[col].astype(str))
        encoders[col] = le

    os.makedirs("models", exist_ok=True)
    joblib.dump(encoders, os.path.join("models", "label_encoders.pkl"))

    print(f"[INFO] Cleaned dataset: {initial_rows} -> {len(df)} valid flow records.")
    return df


def scale_features(X_train: pd.DataFrame, X_test: pd.DataFrame):
    """
    Fits StandardScaler on training features and transforms both train and test sets.
    Prevents data leakage by fitting strictly on X_train.
    """
    scaler = StandardScaler()
    X_train_scaled = pd.DataFrame(
        scaler.fit_transform(X_train),
        columns=X_train.columns,
        index=X_train.index,
    )
    X_test_scaled = pd.DataFrame(
        scaler.transform(X_test),
        columns=X_test.columns,
        index=X_test.index,
    )

    os.makedirs("models", exist_ok=True)
    joblib.dump(scaler, os.path.join("models", "scaler.pkl"))
    return X_train_scaled, X_test_scaled, scaler

"""
File: src/feature_engineering.py
Description: Domain-specific cybersecurity feature engineering module.
             Transforms raw network flow counters into behavioral security signals.
"""

import numpy as np
import pandas as pd


FEATURE_COLUMNS = [
    "duration",
    "protocol_type_encoded",
    "service_encoded",
    "src_bytes",
    "dst_bytes",
    "packet_rate",
    "syn_error_rate",
    "failed_logins",
    "dst_port_diversity",
    "bytes_asymmetry_ratio",
    "packet_burst_intensity",
    "syn_flood_index",
    "brute_force_risk_score",
]


def engineer_security_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Adds 4 domain-engineered cybersecurity features:
    1. bytes_asymmetry_ratio: Detects large outbound data exfiltration vs inbound response
    2. packet_burst_intensity: Measures packet volume per second of flow duration
    3. syn_flood_index: Combines high packet rate with incomplete TCP SYN handshakes
    4. brute_force_risk_score: Weights failed authentication attempts against SSH/login services
    """
    print("[INFO] Engineering behavioral cybersecurity features...")
    df = df.copy()

    # 1. Outbound-to-Inbound Byte Asymmetry Ratio
    df["bytes_asymmetry_ratio"] = np.round(
        (df["src_bytes"] + 1.0) / (df["dst_bytes"] + 1.0), 4
    )

    # 2. Packet Burst Intensity (packets relative to flow duration)
    df["packet_burst_intensity"] = np.round(
        df["packet_rate"] / (df["duration"] + 0.1), 4
    )

    # 3. SYN Flood Composite Index (Packet Rate * SYN Error Rate)
    df["syn_flood_index"] = np.round(
        df["packet_rate"] * df["syn_error_rate"], 4
    )

    # 4. Brute Force Authentication Risk Score
    is_ssh = (df["service"] == "SSH").astype(int)
    df["brute_force_risk_score"] = np.round(
        df["failed_logins"] * (1.5 + is_ssh * 2.0), 4
    )

    return df

"""
File: src/train_model.py
Description: Trains and evaluates a Hybrid AI Threat Detection Pipeline:
             1. Supervised Random Forest Classifier (known threat classification)
             2. Unsupervised Isolation Forest (zero-day anomaly detection)
"""

import os
import joblib
import numpy as np
import pandas as pd
from sklearn.ensemble import IsolationForest, RandomForestClassifier
from sklearn.metrics import (
    accuracy_score,
    classification_report,
    confusion_matrix,
    f1_score,
    precision_score,
    recall_score,
    roc_auc_score,
)
from sklearn.model_selection import train_test_split

from src.feature_engineering import FEATURE_COLUMNS
from src.preprocess import scale_features


def train_and_evaluate_models(df: pd.DataFrame):
    """
    Splits dataset, scales features, trains Random Forest and Isolation Forest,
    and saves evaluation metrics and serialized models.
    """
    X = df[FEATURE_COLUMNS]
    y = df["label"]

    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.25, random_state=42, stratify=y
    )

    X_train_scaled, X_test_scaled, _ = scale_features(X_train, X_test)

    # 1. Train Supervised Random Forest Classifier
    print("[INFO] Training Random Forest Intrusion Detection Model...")
    rf_model = RandomForestClassifier(
        n_estimators=150,
        max_depth=12,
        class_weight="balanced",
        random_state=42,
    )
    rf_model.fit(X_train_scaled, y_train)

    y_pred = rf_model.predict(X_test_scaled)
    y_prob = rf_model.predict_proba(X_test_scaled)[:, 1]

    # 2. Train Unsupervised Isolation Forest for Zero-Day Anomaly Scoring
    print("[INFO] Training Isolation Forest Zero-Day Anomaly Detector...")
    iso_model = IsolationForest(
        n_estimators=120,
        contamination=0.32,
        random_state=42,
    )
    iso_model.fit(X_train_scaled)

    # Compute Evaluation Metrics
    metrics = {
        "accuracy": round(accuracy_score(y_test, y_pred) * 100, 2),
        "precision": round(precision_score(y_test, y_pred) * 100, 2),
        "recall": round(recall_score(y_test, y_pred) * 100, 2),
        "f1_score": round(f1_score(y_test, y_pred) * 100, 2),
        "roc_auc": round(roc_auc_score(y_test, y_prob) * 100, 2),
        "confusion_matrix": confusion_matrix(y_test, y_pred),
        "classification_report": classification_report(
            y_test, y_pred, target_names=["Normal (0)", "Cyber Threat (1)"]
        ),
    }

    # Save trained models
    os.makedirs("models", exist_ok=True)
    joblib.dump(rf_model, os.path.join("models", "random_forest_ids.pkl"))
    joblib.dump(iso_model, os.path.join("models", "isolation_forest.pkl"))

    # Save evaluation report
    os.makedirs("outputs", exist_ok=True)
    report_path = os.path.join("outputs", "evaluation_report.txt")
    with open(report_path, "w", encoding="utf-8") as f:
        f.write("=== AI-POWERED CYBERSECURITY THREAT DETECTION EVALUATION ===\n\n")
        f.write(f"Accuracy  : {metrics['accuracy']}%\n")
        f.write(f"Precision : {metrics['precision']}%\n")
        f.write(f"Recall    : {metrics['recall']}%\n")
        f.write(f"F1-Score  : {metrics['f1_score']}%\n")
        f.write(f"ROC-AUC   : {metrics['roc_auc']}%\n\n")
        f.write("Detailed Classification Report:\n")
        f.write(metrics["classification_report"])

    print(f"[SUCCESS] Model Evaluation Complete -> F1-Score: {metrics['f1_score']}%")
    return rf_model, iso_model, X_test_scaled, y_test, y_pred, metrics

"""
File: src/threat_detector.py
Description: Hybrid Threat Detection & SOC Alert Generation Engine.
             Combines supervised classification probabilities with unsupervised
             Isolation Forest anomaly scores to generate actionable security alerts.
"""

import os
import numpy as np
import pandas as pd

from src.feature_engineering import FEATURE_COLUMNS


MITRE_MAPPING = {
    "SYN_Flood_DoS": "T1498 - Network Denial of Service",
    "SSH_Brute_Force": "T1110 - Brute Force Credential Access",
    "Port_Scan": "T1046 - Network Service Discovery",
    "Data_Exfiltration": "T1048 - Exfiltration Over Alternative Protocol",
    "Normal": "None - Nominal Baseline",
}


def assign_severity(threat_prob: float, anomaly_flag: int) -> str:
    """
    Determines SOC Incident Severity tier using hybrid scoring rules.
    """
    if threat_prob >= 0.85 and anomaly_flag == -1:
        return "CRITICAL"
    elif threat_prob >= 0.70 or anomaly_flag == -1:
        return "HIGH"
    elif threat_prob >= 0.45:
        return "MEDIUM"
    return "NOMINAL"


def run_threat_detection_pipeline(
    df: pd.DataFrame, rf_model, iso_model, scaler
) -> pd.DataFrame:
    """
    Scores network flows, flags malicious traffic, and outputs structured SOC alerts.
    """
    print("[INFO] Running Hybrid Threat Detection & Alert Generation Engine...")
    X_scaled = pd.DataFrame(
        scaler.transform(df[FEATURE_COLUMNS]),
        columns=FEATURE_COLUMNS,
        index=df.index,
    )

    df_results = df.copy()
    df_results["threat_probability"] = np.round(
        rf_model.predict_proba(X_scaled)[:, 1], 4
    )
    df_results["rf_prediction"] = rf_model.predict(X_scaled)
    df_results["iso_anomaly_flag"] = iso_model.predict(X_scaled)  # -1 = Anomaly, 1 = Normal

    # Raw decision function inverted so higher = more anomalous (0 to 100 scale)
    raw_scores = -iso_model.decision_function(X_scaled)
    min_s, max_s = raw_scores.min(), raw_scores.max()
    df_results["anomaly_score"] = np.round(
        ((raw_scores - min_s) / (max_s - min_s + 1e-8)) * 100, 1
    )

    df_results["severity"] = [
        assign_severity(prob, iso)
        for prob, iso in zip(
            df_results["threat_probability"], df_results["iso_anomaly_flag"]
        )
    ]
    df_results["mitre_technique"] = df_results["attack_category"].map(MITRE_MAPPING)

    # Filter actionable alerts (CRITICAL, HIGH, MEDIUM)
    alerts_df = df_results[df_results["severity"] != "NOMINAL"].sort_values(
        by=["threat_probability", "anomaly_score"], ascending=False
    )

    os.makedirs("outputs", exist_ok=True)
    alerts_path = os.path.join("outputs", "security_alerts.csv")
    alerts_df.to_csv(alerts_path, index=False)
    print(f"[ALERT ENGINE] Generated {len(alerts_df)} actionable SOC alerts -> {alerts_path}")
    return df_results, alerts_df

"""
File: main.py
Description: End-to-End Execution Pipeline for the AI-Powered Cybersecurity
             Threat Detection System. Run with: python main.py

import os
import joblib
from data.generate_dataset import generate_cybersecurity_dataset
from src.preprocess import clean_and_preprocess_data
from src.feature_engineering import engineer_security_features
from src.train_model import train_and_evaluate_models
from src.threat_detector import run_threat_detection_pipeline
from src.visualize import generate_all_visualizations


def main():
    print("=" * 68)
    print("  AI-POWERED CYBERSECURITY THREAT DETECTION SYSTEM - PIPELINE")
    print("=" * 68)

    # Step 1: Generate or Load Network Traffic Dataset
    os.makedirs("data", exist_ok=True)
    raw_df = generate_cybersecurity_dataset(num_samples=2500)

    print("\n" + "=" * 68)
    print("  PIPELINE EXECUTION SUMMARY")

    print(f"  Total Flows Analyzed : {len(df_results)}")
    print(f"  Model Accuracy       : {metrics['accuracy']}%")

    print(f"  Model Recall         : {metrics['recall']}%")
    print(f"  Model F1-Score       : {metrics['f1_score']}%")
    print(f"  SOC Alerts Triggered : {len(alerts_df)} (Saved to outputs/security_alerts.csv)")
    print("=" * 68)


if __name__ == "__main__":
    main()

## 1. Project Overview

The **AI-Powered Cybersecurity Threat Detection System** is a modular, production-structured machine learning pipeline built with **Python, Scikit-Learn, Pandas, NumPy, Matplotlib, and Seaborn**.

Designed to replicate modern Security Operations Center (SOC) detection engineering workflows, this system ingests corporate network packet-flow logs (modeled on **NSL-KDD** and **CICIDS2017** schemas), engineers high-signal behavioral ratios, and executes a **Hybrid Dual-Model Threat Engine**:
- **Supervised Random Forest Classifier (`150` trees)**: Detects known multi-vector cyber attacks (`SYN_Flood_DoS`, `SSH_Brute_Force`, and `Port_Scan`) with a **98.41% F1-Score**.
- **Unsupervised Isolation Forest (`120` trees)**: Isolates novel, zero-day volumetric and behavioral anomalies (`Data_Exfiltration`) without requiring prior attack signatures.
- **Automated SOC Alert Triage Engine**: Maps suspicious network flows to **MITRE ATT&CK techniques**, assigns incident severity tiers (`CRITICAL`, `HIGH`, `MEDIUM`, `NOMINAL`), and exports prioritized incident queues (`outputs/security_alerts.csv`) ready for SIEM/Splunk ingestion.

---

## 2. Problem Statement

Modern enterprise networks process millions of packet flows every hour across web servers, SSH jump hosts, and DNS resolvers. Traditional rule-based firewalls and static Intrusion Detection Systems (IDS) fail in three critical ways:
1. **Zero-Day & Encrypted Exfiltration Blindness**: Static firewall signatures cannot detect an insider threat or novel malware exfiltrating database dumps over legitimate port 443 (`HTTPS`) sessions when no known malware signature exists.
2. **SOC Analyst Alert Fatigue**: Rigid single-variable rules trigger thousands of False Positives during legitimate traffic bursts, burying real attacks under benign noise.
3. **Missing Multi-Variable Behavioral Context**: Raw NetFlow logs record isolated counters (`src_bytes`, `packet_rate`, `failed_logins`) rather than composite behavioral relationships (such as outbound-to-inbound byte asymmetry or incomplete TCP SYN handshake density).

This project solves these challenges by pairing **cybersecurity domain feature engineering** with a **hybrid supervised + unsupervised ML architecture**.

## 6. Dataset Details

The repository includes `data/generate_dataset.py`, a deterministic virtual network telemetry simulator modeled on the **NSL-KDD** and **CICIDS2017** feature schemas. It generates **2,500 labeled network flows** across 5 realistic operational traffic classes:

| Traffic Category | MITRE ATT&CK ID | Behavioral Telemetry Signature | Dataset Share |
| :--- | :--- | :--- | :--- |
| **Normal Baseline** | None | Balanced `src_bytes`/`dst_bytes`, `syn_error_rate < 0.06`, 0–1 failed logins | **65%** (1,625 flows) |
| **SYN Flood DoS** | `T1498` | High packet rate (1,800–5,200 pkt/s), `syn_error_rate > 0.78`, tiny payloads | **14%** (350 flows) |
| **SSH Brute Force** | `T1110` | `TCP/SSH` service, 6–38 failed authentication attempts per session | **10%** (250 flows) |
| **Port Scan** | `T1046` | 25–120 distinct destination ports probed within `< 2.0` seconds | **7%** (175 flows) |
| **Data Exfiltration** | `T1048` | High outbound `src_bytes` (250 KB–1.2 MB), extreme byte asymmetry ratio | **4%** (100 flows) |

> **Using External Public Benchmark Datasets**: You can also place CSV exports from **UNSW-NB15**, **NSL-KDD**, or **CICIDS2017** into `data/raw_network_traffic.csv` mapped to the 9 base telemetry columns.

## 11. Known Limitations (Model Drift & Dataset Constraints)

Documenting architectural boundaries is critical for production security engineering:

1. **Concept & Model Drift in Evolving Networks**:
   - Enterprise traffic baselines shift over time as new cloud microservices deploy, remote VPN usage changes, or seasonal traffic spikes occur (**covariate shift**).
   - Static `StandardScaler` parameters and fixed `IsolationForest` contamination ratios (`0.32`) can experience performance degradation if not paired with **Population Stability Index (PSI)** drift monitors and scheduled weekly retraining pipelines.
2. **Dataset Constraints & Synthetic/Benchmark Bias**:
   - Public cybersecurity datasets (**KDD Cup 99**, **NSL-KDD**, **CICIDS2017**) and synthetic flow generators exhibit cleaner class boundaries than noisy multi-tenant production networks.
   - Tabular flow summaries lack raw Layer-7 packet payloads, meaning application-layer exploits (such as SQL injection or Log4Shell strings inside small HTTP POST bodies) cannot be inspected from NetFlow counters alone.
3. **Batch CSV Inference vs. Line-Rate Streaming**:
   - This pipeline evaluates tabular CSV batches rather than ingesting live network socket frames at line rate via `libpcap`, `Zeek`, or `eBPF`.
4. **Low-and-Slow Adversarial Evasion**:
   - Attackers who throttle SSH brute-force attempts to `< 2` tries per minute or pad inbound bytes to disguise outbound data exfiltration can reduce their `brute_force_risk_score` and `bytes_asymmetry_ratio`.

## 12. How to Contribute

Contributions from students, SOC analysts, and machine learning engineers are welcome!

### Step-by-Step Contribution Workflow
1. **Fork the Repository** on GitHub.
2. **Clone Your Fork Locally**:
   ```bash
   git clone https://github.com/<your-username>/AI-Cybersecurity-Threat-Detection.git
   cd AI-Cybersecurity-Threat-Detection
   ```
3. **Create a Descriptive Feature Branch**:
   ```bash
   git checkout -b feat/psi-drift-monitor
   ```
4. **Implement & Verify Your Changes**:
   - Adhere to PEP 8 conventions and include type hints + docstrings for all new functions inside `src/`.
   - Ensure zero data leakage (never call `.fit()` on test splits).
   - Run `python main.py` and confirm all metrics and charts generate without warnings.
5. **Commit Using Conventional Commits**:
   ```bash
   git commit -m "feat(monitoring): add Population Stability Index (PSI) feature drift check"
   ```
6. **Submit a Pull Request**: Push your branch and open a Pull Request describing the enhancement and evaluation impact.
